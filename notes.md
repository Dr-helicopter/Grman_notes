## verbs 
# common verb form
{

    ich         | bin   | brauche   
    du          | bist  | brauchst  
    er/sie/es   | ist   | braucht   
    wir         | sind  | brauchen  
    ihr         | seid  | braucht   
    Sie         | sind  | brauchen  
}

# common verbs
    brauchen    : to need/want
    finden      : to find
    komme       : to come
    haben       : to have


# fucked up verbs
{
    essen       : to eat

    ich         | esse
    du          | isst
    er/sie/es   | isst
    wir         | essen
    ihr         | esst

}
